@version: 4.6

source DataPower_src {
    network(
        ip("0.0.0.0")
        port(514)
        transport("udp")
        flags(no-parse)
    );
};

filter f_DataPower_system_PCC_late_log {
    match("PCC-latency-log" value("MSG"));
};

rewrite r_strip_priority {
    subst(
        "^<[0-9]+>",
        "",
        value("MSG")
        type("pcre")
    );
};

rewrite r_insert_ip {
    subst(
        "^((?:Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec) +[0-9]+ [0-9]{2}:[0-9]{2}:[0-9]{2}) PCC-latency-log",
        "$1 PCC-latency-log/SOURCEIP_PLACEHOLDER",
        value("MSG")
        type("pcre")
    );
};

rewrite r_replace_placeholder {
    set("$(replace-delimiter "SOURCEIP_PLACEHOLDER" "${SOURCEIP}" "${MSG}")", value("MSG"));
};

destination DataPower_system_PCC_late_log {
    file("/var/log/DataPower/PCC/latency.log"
        create_dirs(yes)
        perm(0644)
        dir_perm(0744)
        template("${MSG}\n")
    );
};

log {
    source(DataPower_src);
    filter(f_DataPower_system_PCC_late_log);
    rewrite(r_strip_priority);
    rewrite(r_insert_ip);
    destination(DataPower_system_PCC_late_log);
    flags(final);
};
