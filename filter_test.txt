@version: 4.6

source DataPower_src {
    network(
        ip("0.0.0.0")
        port(514)
        transport("udp")
        # Remove flags(no-parse) — let syslog-ng parse RFC 3164
    );
};

filter f_DataPower_system_PCC_late_log {
    match("PCC-latency-log");
};

destination DataPower_system_PCC_late_log {
    file("/var/log/DataPower/PCC/latency.log"
        create_dirs(yes)
        perm(0644)
        dir_perm(0744)
        template("${DATE} ${HOST} ${FACILITY}.${LEVEL} ${MSGHDR}${MSG}\n")
    );
};

log {
    source(DataPower_src);
    filter(f_DataPower_system_PCC_late_log);
    destination(DataPower_system_PCC_late_log);
    flags(final);
};
