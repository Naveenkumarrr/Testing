cat > /etc/syslog-ng/syslog-ng.conf << 'EOF'
@version: 4.6

source DataPower_src {
    network(
        ip("0.0.0.0")
        port(514)
        transport("udp")
        flags(no-parse)
    );
};

filter f_DataPower_system_PCC_late_log {
    match("PCC-latency-log" value("MSG"));
};

rewrite r_strip_priority {
    subst(
        "^<[0-9]+>",
        "",
        value("MSG")
        type("pcre")
    );
};

rewrite r_insert_ip {
    subst(
        "PCC-latency-log \[",
        "PCC-latency-log/${SOURCEIP} local1.info [",
        value("MSG")
        type("pcre")
    );
};

destination DataPower_system_PCC_late_log {
    file("/var/log/DataPower/PCC/latency.log"
        create_dirs(yes)
        perm(0644)
        dir_perm(0744)
        template("${MSG}\n")
    );
};

log {
    source(DataPower_src);
    filter(f_DataPower_system_PCC_late_log);
    rewrite(r_strip_priority);
    rewrite(r_insert_ip);
    destination(DataPower_system_PCC_late_log);
    flags(final);
};
EOF
