Wait — $(echo) won't work either. The cleanest approach for 4.6 is using $(replace-delimiter) or a Python parser. Let me give you the guaranteed working solution using a Python parser:
bashcat > /etc/syslog-ng/syslog-ng.conf << 'EOF'
@version: 4.6

python {
import re

class DataPowerParser(object):
    def init(self, options):
        return True

    def parse(self, log_message):
        msg = log_message['MSG'].decode('utf-8', errors='replace')
        sourceip = log_message['SOURCEIP'].decode('utf-8', errors='replace')
        msg = re.sub(r'^<[0-9]+>', '', msg)
        msg = re.sub(
            r'^([A-Za-z]+ +[0-9]+ [0-9]{2}:[0-9]{2}:[0-9]{2}) PCC-latency-log ',
            r'\1 PCC-latency-log/' + sourceip + ' local1.info ',
            msg
        )
        log_message['MSG'] = msg
        return True
};

source DataPower_src {
    network(
        ip("0.0.0.0")
        port(514)
        transport("udp")
        flags(no-parse)
    );
};

filter f_DataPower_system_PCC_late_log {
    match("PCC-latency-log" value("MSG"));
};

parser p_datapower {
    python(
        class("DataPowerParser")
    );
};

destination DataPower_system_PCC_late_log {
    file("/var/log/DataPower/PCC/latency.log"
        create_dirs(yes)
        perm(0644)
        dir_perm(0744)
        template("${MSG}\n")
    );
};

log {
    source(DataPower_src);
    filter(f_DataPower_system_PCC_late_log);
    parser(p_datapower);
    destination(DataPower_system_PCC_late_log);
    flags(final);
};
